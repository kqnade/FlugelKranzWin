cmake_minimum_required(VERSION 3.20)
project(FlugelKranzDriver LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/minhook)
add_library(driver_flugelkranz SHARED Driver.cpp)
target_include_directories(driver_flugelkranz PRIVATE vendor/openvr)
target_link_libraries(driver_flugelkranz PRIVATE minhook)
add_executable(transform_tests TransformTests.cpp)
target_include_directories(transform_tests PRIVATE vendor/openvr)
enable_testing()
add_test(NAME transforms COMMAND transform_tests)
add_executable(observer_tests ObserverTests.cpp)
target_include_directories(observer_tests PRIVATE vendor/openvr)
target_link_libraries(observer_tests PRIVATE minhook)
add_test(NAME observer_forwarding COMMAND observer_tests)
add_executable(frame_tests FrameTests.cpp)
target_include_directories(frame_tests PRIVATE vendor/openvr)
add_test(NAME frame_history COMMAND frame_tests)
